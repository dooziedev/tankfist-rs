rule("Toggle recording (Interact)")
{
	event
	{
		Ongoing - Each Player;
		All;
		Doomfist;
	}

	conditions
	{
		Is Dummy Bot(Event Player) == False;
		Is Button Held(Event Player, Button(Interact)) == True;
		Is Button Held(Event Player, Button(Crouch)) == False;
		Is Button Held(Event Player, Button(Melee)) == False;
		Is Button Held(Event Player, Button(Ultimate)) == False;
		Is Button Held(Event Player, Button(Reload)) == False;
		Is Button Held(Event Player, Button(Primary Fire)) == False;
	}

	actions
	{
		Global.Recording = !Global.Recording;
		If(Global.Recording == True);
			Global.Take += 1;
			Event Player.Origin = Position Of(Event Player);
			Event Player.OriginFacing = Facing Direction Of(Event Player);
		End;
		Log To Inspector(Custom String("R;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Take, Global.Recording));
		Small Message(Event Player, Custom String("Take {0} recording: {1}", Global.Take, Global.Recording));
	}
}

rule("Run bunny-hop bot tests (Primary Fire + Interact)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is Dummy Bot(Event Player) == False;
		Is Button Held(Event Player, Button(Primary Fire)) == True;
		Is Button Held(Event Player, Button(Interact)) == True;
		Global.Running == False;
		Global.Recording == False;
	}

	actions
	{
		Global.Running = True;
		Global.Yaw = Horizontal Facing Angle Of(Event Player);
		Global.BotOrigin = Position Of(Event Player) + Direction From Angles(Global.Yaw - 90, 0) * 4;
		If(Entity Exists(Global.Bot) == False);
			Create Dummy Bot(Hero(Doomfist), Team Of(Event Player), -1, Global.BotOrigin, Direction From Angles(Global.Yaw, 0));
			Wait(1, Ignore Condition);
			Global.Bot = First Of(Filtered Array(All Players(Team Of(Event Player)), Is Dummy Bot(Current Array Element) == True && Hero Of(Current Array Element) == Hero(Doomfist)));
		End;
		Small Message(All Players(All Teams), Custom String("Bunny-hop tests running"));
		"Suite 10: jump held through touchdown"
		Global.Suite = 10;
		Global.Mode = 0;
		For Global Variable(J, 0, Count Of(Global.Pitches), 1);
			Global.Pitch = Value In Array(Global.Pitches, Global.J);
			For Global Variable(I, 0, Count Of(Global.HoldStarts), 1);
				Global.Delay = Value In Array(Global.HoldStarts, Global.I);
				Call Subroutine(RunBhop);
			End;
		End;
		Global.Pitch = 0;
		Global.Delay = 0.1;
		Global.Mode = 1;
		Call Subroutine(RunBhop);
		Global.Mode = 0;
		"Suite 11: jump pressed late, after touching down"
		Global.Suite = 11;
		For Global Variable(I, 0, Count Of(Global.LateDelays), 1);
			Global.Delay = Value In Array(Global.LateDelays, Global.I);
			Call Subroutine(RunBhop);
		End;
		"Suite 12: punch, jump cancel, slam, jump held through touchdown"
		Global.Suite = 12;
		Global.Delay = 0.1;
		For Global Variable(J, 0, 2, 1);
			Global.Pitch = Value In Array(Global.Pitches, Global.J);
			Call Subroutine(RunBhop);
		End;
		Small Message(All Players(All Teams), Custom String("Bunny-hop tests done"));
		Global.Running = False;
	}
}

rule("Sub: run one bunny-hop test")
{
	event
	{
		Subroutine;
		RunBhop;
	}

	actions
	{
		Stop Throttle In Direction(Global.Bot);
		Stop Holding Button(Global.Bot, Button(Jump));
		Stop Holding Button(Global.Bot, Button(Secondary Fire));
		Wait(0.3, Ignore Condition);
		Teleport(Global.Bot, Global.BotOrigin);
		Set Facing(Global.Bot, Direction From Angles(Global.Yaw, Global.Pitch), To World);
		Set Ability Cooldown(Global.Bot, Button(Secondary Fire), 0);
		Set Ability Cooldown(Global.Bot, Button(Ability 1), 0);
		Heal(Global.Bot, Null, 10000);
		Wait(0.7, Ignore Condition);
		Global.Take += 1;
		Global.BotRec = True;
		Log To Inspector(Custom String("R;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Take, True));
		Log To Inspector(Custom String("X;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Suite, Custom String("{0};{1};{2}", Round To Integer(Global.Pitch * 1000, To Nearest), Round To Integer(Global.Delay * 1000, To Nearest), Global.Mode)));
		Wait(0.2, Ignore Condition);
		If(Global.Mode == 1);
			Start Throttle In Direction(Global.Bot, Forward, 1, To Player, Replace Existing Throttle, Direction and Magnitude);
			Wait(0.6, Ignore Condition);
		End;
		If(Global.Suite == 12);
			Start Holding Button(Global.Bot, Button(Secondary Fire));
			Log To Inspector(Custom String("E;{0};4", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(1, Ignore Condition);
			Stop Holding Button(Global.Bot, Button(Secondary Fire));
			Log To Inspector(Custom String("E;{0};5", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.2, Ignore Condition);
			Press Button(Global.Bot, Button(Jump));
			Log To Inspector(Custom String("E;{0};3", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.1, Ignore Condition);
		End;
		Press Button(Global.Bot, Button(Ability 1));
		Log To Inspector(Custom String("E;{0};1", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		If(Global.Suite == 11);
			Wait(0.15, Ignore Condition);
			Wait Until(Is On Ground(Global.Bot), 3);
			Log To Inspector(Custom String("E;{0};9", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(Max(Global.Delay, 0.016), Ignore Condition);
			Press Button(Global.Bot, Button(Jump));
			Log To Inspector(Custom String("E;{0};3", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		Else;
			Wait(Global.Delay, Ignore Condition);
			Start Holding Button(Global.Bot, Button(Jump));
			Log To Inspector(Custom String("E;{0};7", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.15, Ignore Condition);
			Wait Until(Is On Ground(Global.Bot), 3);
			Log To Inspector(Custom String("E;{0};9", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.5, Ignore Condition);
			Stop Holding Button(Global.Bot, Button(Jump));
			Log To Inspector(Custom String("E;{0};8", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		End;
		Wait(1.5, Ignore Condition);
		Stop Throttle In Direction(Global.Bot);
		Stop Holding Button(Global.Bot, Button(Jump));
		Global.BotRec = False;
		Log To Inspector(Custom String("R;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Take, False));
	}
}
