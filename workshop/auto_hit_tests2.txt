rule("Setup: test lists")
{
	event
	{
		Ongoing - Global;
	}

	actions
	{
		"Suite 30: attacker angle around the blocking bot, degrees"
		Global.Angles = Array(30, 35, 40, 45, 50, 55, 60);
		"Suite 31: one frontal hit of this raw damage while blocking (prevented = 75%: 94 to 110)"
		Global.Amounts = Array(125.34, 128, 130.67, 133.34, 136, 138.67, 141.34, 144, 146.67);
		"Suite 32: slam target offsets from the landing point (ahead, sideways), 4 per grid"
		Global.GridF = Array(4, 4, 4, 4, 4, 4, 4, 4, 0.5, 1, 1.5, 9, 10, 12, 14, -0.5);
		Global.GridS = Array(0.5, 1, 1.5, 2, 2.5, 3, -1, -1.5, 0, 0, 0, 0, 0, 0, 0, 0);
		"Suite 33: wall slam range: (charge, distance from the wall)"
		Global.Charges = Array(1, 1, 1, 0, 0);
		Global.WallDistances = Array(16, 20, 24, 8, 12);
		"Suite 34: empowered splash: second target this far to the side of the main one, m"
		Global.Hits = Array(2, 3, 4, 5, 6);
	}
}

rule("Run all tests (Interact)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is Dummy Bot(Event Player) == False;
		Is Button Held(Event Player, Button(Interact)) == True;
		Global.Running == False;
	}

	actions
	{
		Global.Running = True;
		Global.Yaw = Horizontal Facing Angle Of(Event Player);
		Global.Fwd = Direction From Angles(Global.Yaw, 0);
		Global.Side = Direction From Angles(Global.Yaw - 90, 0);
		Global.Origin = Position Of(Event Player) + Global.Side * 4;
		Global.WallDist = Distance Between(Eye Position(Event Player), Ray Cast Hit Position(Eye Position(Event Player), Eye Position(Event Player) + Global.Fwd * 80, Null, All Players(All Teams), False));
		Global.Landing = Global.Origin + Global.Fwd * 13.81;
		If(Entity Exists(Global.Bot) == False);
			Create Dummy Bot(Hero(Doomfist), Team Of(Event Player), -1, Global.Origin, Global.Fwd);
			Create Dummy Bot(Hero(Ana), Opposite Team Of(Team Of(Event Player)), 0, Global.Origin - Global.Fwd * 12, Global.Fwd);
			Create Dummy Bot(Hero(Ana), Opposite Team Of(Team Of(Event Player)), 1, Global.Origin - Global.Fwd * 12 + Global.Side * 3, Global.Fwd);
			Create Dummy Bot(Hero(Ana), Opposite Team Of(Team Of(Event Player)), 2, Global.Origin - Global.Fwd * 12 + Global.Side * 6, Global.Fwd);
			Create Dummy Bot(Hero(Ana), Opposite Team Of(Team Of(Event Player)), 3, Global.Origin - Global.Fwd * 12 + Global.Side * 9, Global.Fwd);
			Create Dummy Bot(Hero(Reinhardt), Opposite Team Of(Team Of(Event Player)), 4, Global.Origin - Global.Fwd * 12 + Global.Side * 12, Global.Fwd);
			Create Dummy Bot(Hero(Soldier: 76), Opposite Team Of(Team Of(Event Player)), 5, Global.Origin - Global.Fwd * 12 + Global.Side * 15, Global.Fwd);
			Wait(1.5, Ignore Condition);
			Global.Bot = First Of(Filtered Array(All Players(Team Of(Event Player)), Is Dummy Bot(Current Array Element) == True && Hero Of(Current Array Element) == Hero(Doomfist)));
			Global.Targets = Array(Players In Slot(0, Opposite Team Of(Team Of(Event Player))), Players In Slot(1, Opposite Team Of(Team Of(Event Player))), Players In Slot(2, Opposite Team Of(Team Of(Event Player))), Players In Slot(3, Opposite Team Of(Team Of(Event Player))), Players In Slot(4, Opposite Team Of(Team Of(Event Player))));
			Global.Attacker = Players In Slot(5, Opposite Team Of(Team Of(Event Player)));
			Set Max Health(Global.Targets, 1000);
			Set Max Health(Global.Attacker, 1000);
		End;
		Small Message(All Players(All Teams), Custom String("Hit tests running (wall {0} m)", Round To Integer(Global.WallDist, To Nearest)));

		"Suite 30: Power Block arc. LabelA = angle, LabelC = raw damage"
		Global.Suite = 30;
		For Global Variable(I, 0, Count Of(Global.Angles), 1);
			Global.LabelA = Value In Array(Global.Angles, Global.I);
			Global.LabelB = 1;
			Global.LabelC = 100;
			Call Subroutine(ResetAll);
			Teleport(Global.Attacker, Global.Origin + Direction From Angles(Global.Yaw + Global.LabelA, 0) * 8);
			Wait(0.3, Ignore Condition);
			Call Subroutine(BeginTake);
			Press Button(Global.Bot, Button(Ability 2));
			Log To Inspector(Custom String("E;{0};2", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.4, Ignore Condition);
			Damage(Global.Bot, Global.Attacker, 100);
			Log To Inspector(Custom String("E;{0};11", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Call Subroutine(FinishTake);
		End;

		"Suite 31: empower threshold, one frontal hit. LabelA = raw damage, then a full punch at a target"
		Global.Suite = 31;
		For Global Variable(I, 0, Count Of(Global.Amounts), 1);
			Global.LabelA = Value In Array(Global.Amounts, Global.I);
			Global.LabelB = 0;
			Global.LabelC = 0;
			Call Subroutine(ResetAll);
			Teleport(Global.Attacker, Global.Origin + Global.Fwd * 8);
			Wait(0.3, Ignore Condition);
			Call Subroutine(BeginTake);
			Press Button(Global.Bot, Button(Ability 2));
			Log To Inspector(Custom String("E;{0};2", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.4, Ignore Condition);
			Damage(Global.Bot, Global.Attacker, Global.LabelA);
			Log To Inspector(Custom String("E;{0};11", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait Until(Is Using Ability 2(Global.Bot) == False, 4);
			Log To Inspector(Custom String("E;{0};12", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Heal(Global.Bot, Null, 10000);
			Teleport(Global.Attacker, Global.Origin - Global.Fwd * 12 + Global.Side * 15);
			Teleport(First Of(Global.Targets), Position Of(Global.Bot) + Global.Fwd * 6);
			Wait(0.3, Ignore Condition);
			Global.Charge = 1;
			Call Subroutine(DoPunch);
			Call Subroutine(FinishTake);
		End;

		"Suite 32: slam hit area edges. LabelA = grid number"
		Global.Suite = 32;
		For Global Variable(I, 0, 4, 1);
			Global.LabelA = Global.I;
			Global.LabelB = 0;
			Global.LabelC = 0;
			Call Subroutine(ResetAll);
			For Global Variable(J, 0, 4, 1);
				Teleport(Value In Array(Global.Targets, Global.J), Global.Landing + Global.Fwd * Value In Array(Global.GridF, Global.I * 4 + Global.J) + Global.Side * Value In Array(Global.GridS, Global.I * 4 + Global.J));
			End;
			Wait(0.3, Ignore Condition);
			Call Subroutine(BeginTake);
			Press Button(Global.Bot, Button(Ability 1));
			Log To Inspector(Custom String("E;{0};1", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Call Subroutine(FinishTake);
		End;

		"Suite 33: wall slam range. LabelA = charge s, LabelB = target distance from the wall, LabelC = wall distance"
		Global.Suite = 33;
		For Global Variable(I, 0, Count Of(Global.Charges), 1);
			Global.Charge = Value In Array(Global.Charges, Global.I);
			Global.LabelA = Global.Charge;
			Global.LabelB = Value In Array(Global.WallDistances, Global.I);
			Global.LabelC = Global.WallDist;
			Call Subroutine(ResetAll);
			Teleport(Global.Bot, Global.Origin + Global.Fwd * (Global.WallDist - Global.LabelB - 6));
			Teleport(First Of(Global.Targets), Global.Origin + Global.Fwd * (Global.WallDist - Global.LabelB));
			Wait(0.3, Ignore Condition);
			Call Subroutine(BeginTake);
			Call Subroutine(DoPunch);
			Global.Charge = 1;
			Wait(1, Ignore Condition);
			Call Subroutine(FinishTake);
		End;

		"Suite 34: empowered splash radius. LabelA = sideways distance of the second target, m"
		Global.Suite = 34;
		For Global Variable(I, 0, 5, 1);
			Global.LabelA = Value In Array(Global.Hits, Global.I);
			Global.LabelB = 1;
			Global.LabelC = 0;
			Call Subroutine(ResetAll);
			Call Subroutine(BeginTake);
			Global.LabelC = Global.Hits;
			Global.Hits = 12;
			Call Subroutine(EmpowerBot);
			Global.Hits = Global.LabelC;
			Global.LabelC = 0;
			Teleport(First Of(Global.Targets), Position Of(Global.Bot) + Global.Fwd * 6);
			Teleport(Value In Array(Global.Targets, 1), Position Of(Global.Bot) + Global.Fwd * 6 + Global.Side * Global.LabelA);
			Wait(0.2, Ignore Condition);
			Global.Charge = 1;
			Call Subroutine(DoPunch);
			Call Subroutine(FinishTake);
		End;
		Call Subroutine(ResetAll);
		Small Message(All Players(All Teams), Custom String("Hit tests part 2 done: {0} takes", Global.Take));
		Global.Running = False;
	}
}

rule("Sub: reset everyone")
{
	event
	{
		Subroutine;
		ResetAll;
	}

	actions
	{
		Stop Holding Button(Global.Bot, Button(Secondary Fire));
		Stop Holding Button(Global.Bot, Button(Ability 2));
		Wait(0.3, Ignore Condition);
		Teleport(Global.Bot, Global.Origin);
		Set Facing(Global.Bot, Global.Fwd, To World);
		Set Ability Cooldown(Global.Bot, Button(Secondary Fire), 0);
		Set Ability Cooldown(Global.Bot, Button(Ability 1), 0);
		Set Ability Cooldown(Global.Bot, Button(Ability 2), 0);
		Heal(Global.Bot, Null, 10000);
		For Global Variable(J, 0, 5, 1);
			Teleport(Value In Array(Global.Targets, Global.J), Global.Origin - Global.Fwd * 12 + Global.Side * (Global.J * 3));
			Set Facing(Value In Array(Global.Targets, Global.J), Global.Fwd * -1, To World);
			Heal(Value In Array(Global.Targets, Global.J), Null, 100000);
		End;
		Teleport(Global.Attacker, Global.Origin - Global.Fwd * 12 + Global.Side * 15);
		Heal(Global.Attacker, Null, 100000);
		Wait(0.7, Ignore Condition);
	}
}

rule("Sub: begin a take")
{
	event
	{
		Subroutine;
		BeginTake;
	}

	actions
	{
		Global.Take += 1;
		Global.Recording = True;
		Log To Inspector(Custom String("R;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Take, True));
		Log To Inspector(Custom String("Z;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Suite, Custom String("{0};{1};{2}", Round To Integer(Global.LabelA * 1000, To Nearest), Round To Integer(Global.LabelB * 1000, To Nearest), Round To Integer(Global.LabelC * 1000, To Nearest))));
		Wait(0.2, Ignore Condition);
	}
}

rule("Sub: finish a take")
{
	event
	{
		Subroutine;
		FinishTake;
	}

	actions
	{
		Wait(2.5, Ignore Condition);
		Stop Holding Button(Global.Bot, Button(Secondary Fire));
		Global.Recording = False;
		Log To Inspector(Custom String("R;{0};{1};{2}", Round To Integer(Total Time Elapsed * 1000, To Nearest), Global.Take, False));
	}
}

rule("Sub: punch with Global.Charge seconds of hold")
{
	event
	{
		Subroutine;
		DoPunch;
	}

	actions
	{
		Start Holding Button(Global.Bot, Button(Secondary Fire));
		Log To Inspector(Custom String("E;{0};4", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		Wait(Max(Global.Charge, 0.016), Ignore Condition);
		Stop Holding Button(Global.Bot, Button(Secondary Fire));
		Log To Inspector(Custom String("E;{0};5", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
	}
}

rule("Sub: block, take Global.Hits x 25 frontal damage, wait for the block to end")
{
	event
	{
		Subroutine;
		EmpowerBot;
	}

	actions
	{
		Teleport(Global.Attacker, Position Of(Global.Bot) + Global.Fwd * 8);
		Wait(0.2, Ignore Condition);
		Press Button(Global.Bot, Button(Ability 2));
		Log To Inspector(Custom String("E;{0};2", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		Wait(0.3, Ignore Condition);
		For Global Variable(J, 0, Global.Hits, 1);
			Damage(Global.Bot, Global.Attacker, 25);
			Log To Inspector(Custom String("E;{0};11", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
			Wait(0.05, Ignore Condition);
		End;
		Wait Until(Is Using Ability 2(Global.Bot) == False, 4);
		Log To Inspector(Custom String("E;{0};12", Round To Integer(Total Time Elapsed * 1000, To Nearest)));
		Heal(Global.Bot, Null, 10000);
		Teleport(Global.Attacker, Global.Origin - Global.Fwd * 12 + Global.Side * 15);
		Wait(0.3, Ignore Condition);
	}
}
